# --- Core FL + tooling (CPU is fine for --dry-run / --baseline) ---
# torch>=2.7 is the floor for NVIDIA Blackwell / RTX 50-series (sm_120) GPUs.
# On those cards install the CUDA 12.8 build, e.g.:
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu128
# (cu121/cu124 wheels have no sm_120 kernels and fail on an RTX 5090.)
torch>=2.7
torchvision
numpy
pyyaml
matplotlib
openai>=1.0          # OpenAI inference/baseline backend
requests             # Ollama HTTP backend

# --- RL training stack (install on the CUDA GPU machine) ---
# Unsloth provides 4-bit QLoRA loading + GRPO-friendly Llama 3.2 support and pulls
# in transformers/accelerate/bitsandbytes. GRPO itself is implemented directly
# in rl/grpo.py on top of PEFT adapters (no TRL trainer dependency required).
# NOTE: unsloth >= 2026.6.9 is required — 2026.3.11 has a broken Llama fast
# KV-cache inference kernel (RoPE cos/sin broadcast crash) vs Transformers 5.3.
unsloth>=2026.6.9
unsloth_zoo>=2026.6.7
peft>=0.13
transformers>=4.45
accelerate
bitsandbytes
matplotlib
